#!/usr/bin/env python3
"""LAB FRAME vs WAVE FRAME (2026-10-05, owner: "vedere se sono equivalenti"). Same design, mesh, sector states, far
field, scheme and binary; the lab run (labframe_case.py, BC 410 turning in time) against the wave-frame run
(waveframe_case.py, rotating frame, steady). Three readings:
  1. the wall thrust C_F integrated around the annulus: lab per dump (post_labframe.py) against the wave run's dumps
     (post_waveframe.py); the lab mean over the dumps after T_SKIP (whole periods); the 12 sector thrusts;
  2. the probes: a fixed laboratory probe at the azimuth theta_p sees, once periodic, the wave-frame wall pressure at
     theta_w = theta_p + Omega t (wave toward decreasing theta) -- read from the wave field at the same cell row;
  3. the whole pressure field of the last lab dumps turned into the wave frame against the wave field, per block family
     (Nozzle channel, Outer, Ambient): median / 95th percentile / max of |p_lab - p_wave| / p_ref.
Pass line declared BEFORE reading (TWIN protocol A-REPR): |C_F(lab, periodic mean) - C_F(wave)| < 0.17 % of C_F
(one third of the 0.5 % thrust-stand band).
usage: compare_lab_wave.py <lab case dir> <wave field npz (extract_p.py)> <wave wf_post.json> [--tskip 3e-4]"""
import argparse, glob, json, os
import numpy as np

ap = argparse.ArgumentParser()
ap.add_argument("lab"); ap.add_argument("wave_npz"); ap.add_argument("wave_post"); ap.add_argument("--tskip", type=float, default=3.0e-4)
a = ap.parse_args()
C = json.load(open(os.path.join(a.lab, "case.json")))
S = json.load(open(C["mesh"].replace(".dat", "_summary.json")))
T, F = float(C["period"]), float(C["f_wave"])
fam = {int(b): v["name"].split("-")[0] for b, v in S["blocks"].items()}


def load(f):
    d = np.load(f)
    out = {}
    for b in range(1, int(d["nblocks"][0]) + 1):
        ni, nj, nk = (int(v) for v in d["b%d_dims" % b])
        nn = ni * nj * nk
        xyz = d["b%d_xyz" % b]
        Y = xyz[nn:2 * nn].reshape((ni, nj, nk), order="F"); Z = xyz[2 * nn:].reshape((ni, nj, nk), order="F")
        th = np.unwrap(np.arctan2(Z[0, 0, :], Y[0, 0, :]))
        out[b] = (np.degrees(0.5 * (th[1:] + th[:-1])) % 360.0, d["b%d_p" % b].reshape((ni - 1, nj - 1, nk - 1), order="F"))
    return out, float(d["t"][0])


def annulus(D, family):
    """(theta (ncol,), p (ni, nj, ncol)) of one block family, columns sorted by azimuth."""
    th = np.concatenate([D[b][0] for b in sorted(D) if fam[b] == family])
    p = np.concatenate([D[b][1] for b in sorted(D) if fam[b] == family], axis=2)
    o = np.argsort(th)
    return th[o], p[:, :, o]


def at_theta(th, p, q):
    """p (..., ncol) at the azimuths q (deg), periodic linear interpolation."""
    xp = np.concatenate([[th[-1] - 360.0], th, [th[0] + 360.0]])
    fp = np.concatenate([p[..., -1:], p, p[..., :1]], axis=-1)
    q = np.mod(q, 360.0)
    i = np.clip(np.searchsorted(xp, q) - 1, 0, len(xp) - 2)
    w = (q - xp[i]) / (xp[i + 1] - xp[i])
    return fp[..., i] * (1 - w) + fp[..., i + 1] * w


res = {}
# ---- 1. thrust
lab = json.load(open(os.path.join(a.lab, "lab_post.json")))["dumps"]
wav = json.load(open(a.wave_post))["dumps"]
cf_w = np.array([r["CF"] for r in wav])
late = [r for r in lab if r["t"] >= a.tskip]
# the lab C_F with the inlet term at its period mean (post_labframe.py: a dump catches the 410 table at an arbitrary phase of
# its sample interval; the walls' pushes are flat in time); the instantaneous value is kept for the record
cf_l = np.array([r.get("CF_inlet_mean", r["CF"]) for r in late])
cf_inst = np.array([r["CF"] for r in late])
walls_w = -wav[-1]["push_plug"] + wav[-1]["push_cowl"]
walls_l = np.array([-r["push_plug"] + r["push_cowl"] for r in late])
res["thrust"] = dict(wave_dumps=[(r["t"], r["CF"]) for r in wav], lab_dumps=[(r["t"], r["CF"]) for r in lab],
                     wave_CF=float(cf_w[-1]), lab_CF_mean=float(cf_l.mean()), lab_CF_min=float(cf_l.min()), lab_CF_max=float(cf_l.max()),
                     lab_n=len(late), diff_rel=float(cf_l.mean() / cf_w[-1] - 1.0), pass_line=0.0017,
                     lab_CF_instant_mean=float(cf_inst.mean()), inlet_term_lab=float(late[-1].get("F_in_sample_mean", np.nan)),
                     walls_diff_of_CF=float((walls_l.mean() - walls_w) / cf_w[-1]),
                     PASS=bool(abs(cf_l.mean() / cf_w[-1] - 1.0) < 0.0017))
sw = wav[-1]["sector"]
res["sectors"] = {k: dict(wave=sw[k], lab_mean=float(np.mean([r["sector"][k] for r in late])),
                          rel=float(np.mean([r["sector"][k] for r in late]) / sw[k] - 1.0)) for k in sw}
for k in ("push_plug", "push_cowl"):
    res["thrust"][k] = dict(wave=wav[-1][k], lab_mean=float(np.mean([r[k] for r in late])))
print("C_F wave %.6f | lab mean of %d dumps after %.2e s %.6f (min %.6f max %.6f) -> %+.4f %%  [pass line 0.17 %%: %s]"
      % (cf_w[-1], len(late), a.tskip, cf_l.mean(), cf_l.min(), cf_l.max(), 100 * res["thrust"]["diff_rel"], "PASS" if res["thrust"]["PASS"] else "FAIL"))
print("  of which the walls (plug + cowl pushes): %+.4f %% of C_F; the rest = the inlet term (the 410 table's time sampling)"
      % (100 * res["thrust"]["walls_diff_of_CF"]))
print("sectors (lab/wave - 1, %):", " ".join("%s:%+.3f" % (k, 100 * v["rel"]) for k, v in res["sectors"].items()))
# ---- 2. probes
W, tw = load(a.wave_npz)
pr = {}
for nm, b, i, j, k in C["probes"]:
    f = os.path.join(a.lab, "OUTPUT", nm + ".txt")
    if not os.path.exists(f) or os.path.getsize(f) == 0:
        continue
    d = np.loadtxt(f)
    d = d[d[:, 0] >= a.tskip]
    thp = W[b][0][k - 1]                                             # the probe's laboratory azimuth (same mesh)
    th, p = annulus(W, fam[b])
    pw = at_theta(th, p[i - 1, j - 1, :], thp + 360.0 * F * d[:, 0])
    e = (d[:, 1] - pw) / np.maximum(np.abs(pw), 1.0)
    pr[nm] = dict(theta_p=float(thp), n=int(len(d)), mean_lab=float(d[:, 1].mean()), mean_wave=float(pw.mean()),
                  rel_mean=float(d[:, 1].mean() / pw.mean() - 1.0), rel_median_abs=float(np.median(np.abs(e))), rel_p95_abs=float(np.percentile(np.abs(e), 95)))
    print("probe %-13s theta %6.1f: mean lab %.4e wave %.4e (%+.3f %%), |e| median %.3f %% p95 %.3f %%"
          % (nm, thp, pr[nm]["mean_lab"], pr[nm]["mean_wave"], 100 * pr[nm]["rel_mean"], 100 * pr[nm]["rel_median_abs"], 100 * pr[nm]["rel_p95_abs"]))
res["probes"] = pr
# ---- 3. whole field, the last two lab dumps turned into the wave frame
fields = sorted(glob.glob(os.path.join(a.lab, "OUTPUT", "field*_p.npz")), key=lambda s: (len(s), s))[-2:]
res["field"] = {}
for f in fields:
    L, tl = load(f)
    rowf = {}
    for family in ("Nozzle", "Outer", "Ambient"):
        thw, pw = annulus(W, family)
        thl, pl = annulus(L, family)
        pl_w = at_theta(thl, pl, thw - 360.0 * F * tl)               # the lab field at theta_lab = theta_w - Omega t
        ref = np.maximum(np.abs(pw), 1.0)
        e = np.abs(pl_w - pw) / ref
        rowf[family] = dict(median=float(np.median(e)), p95=float(np.percentile(e, 95)), max=float(e.max()))
    res["field"][os.path.basename(f)] = dict(t=tl, turns=tl / T, **rowf)
    print("field %s t %.3e (%.2f turns): " % (os.path.basename(f), tl, tl / T)
          + " | ".join("%s median %.2e p95 %.2e max %.2e" % (k, v["median"], v["p95"], v["max"]) for k, v in rowf.items()))
json.dump(res, open(os.path.join(a.lab, "compare_lab_wave.json"), "w"), indent=1)
print("written", os.path.join(a.lab, "compare_lab_wave.json"))
