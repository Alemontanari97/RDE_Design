#!/usr/bin/env python3
"""NOZZLE 2026-09-30, THE WAVE-FRAME PILOT (PILOT-WF): the 3D nozzle + exhaust solved by MOSE_open in the frame that
turns with the detonation wave (single rotating frame, rigid body). In that frame the inlet pattern stands still, so the
run is a march to a STEADY flow; its wall thrust, integrated around the annulus, is the cycle-averaged thrust the
per-phase runs estimate one frozen sector at a time.
Inlet modes:
  phases   the 12 sector states of phases_405.json, each on its own 30-degree sector (phase k on [180+30k, 210+30k) deg),
           axial in the laboratory, no swirl: the SAME inlet data as the 12 per-phase wedge runs;
  uniform  the mean state (index 12) all around: the control -- it must give the wedge run of the mean state.
Frame: the wave travels toward decreasing theta at F_WAVE (4246 Hz), so the frame's angular velocity is
-omega x^ (omega > 0 about the axis (-1, 0, 0)); gas at rest in the laboratory moves at +omega r in theta in the frame.
The 405 inlet takes RELATIVE data: Mach |w|/a, static T and p, and the direction angles (radians)
alpha = atan2(w_y, w_x), beta = atan2(w_z, hypot(w_x, w_y)), with w = (u, omega r (-sin th), omega r cos th).
Far field: 406 (back pressure p_a, the rest extrapolated), NOT 407: the still air moves at relative Mach 8 at the outer
boundary and 407, an inflow at relative stagnation conditions, stagnated it there (both first runs of 2026-09-30 died
within 0.2 ms with 400 bar at the outer row).
Builds CASE_3D/wf_<design>_<mode>/ (mesh link, bc, ic in relative velocities, the chemistry kit, input.ini). NOT launched.
usage: waveframe_case.py <mesh.dat> <design> <phases|uniform> [--pamb 5000] [--tamb 290] [--tend 3e-3] [--dumps 12] [--cfl 0.4]"""
import argparse
import json
import os
import subprocess
import sys

import numpy as np

HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, "/data10/falco/RDE/codes/MOSE_GPU/tools")
import tecmesh                                                  # noqa: E402

PY = "/data10/falco/RDE/RDE_Design-rde-nozzle-program/.venv-a1/bin/python"
sys.path.insert(0, HERE)
import kit as KITMOD                                            # noqa: E402
KIT = KITMOD.Kit()
RU = 8314.462618
WSP, AIR = KIT.W, KIT.air()
TURB = (1.0e-6, 1.0e2)
F_WAVE = float(os.environ.get("WF_FREQ", 4246.0))               # Hz: the Q2D THOR_CAv3 wave (period 235.5 us) unless WF_FREQ says otherwise (the methane case: D_CJ over the mid-annulus circumference)
OMEGA = 2.0 * np.pi * F_WAVE
INI = """[MOSE-Parameters]
iter-threshold = 2000000
time-threshold = {tend}
newrun = false
[MOSE-IO]
sol-overwrite = false
sol-diter = 100000000
sol-dtime = {sol_dtime}
shell-diter = 500
sol-format = tecplot ascii
[MOSE-Physics]
equations = euler
chemistry = frozen
rotational-frame = rigid-body
[MOSE-Rotating-Frame]
omega = {omega:.6f}
axis = -1.0 0.0 0.0
origin = 0.0 0.0 0.0
[MOSE-Numerics]
cfl = {cfl}
cfl-rise-threshold = 100
time-accurate = true
time-scheme = RK2
irs-beta = 0.0
integration-variables = cons
riemann-solver = HLLC
space-reconstruction = MUSCL
flux-limiter = minmod
[GPB-Phase1]
phase = ONERA-7
Tmax = 5000
"""


NSEC = 12


def phase_of(theta_deg, mode):
    """The inlet-state index at the azimuth theta (deg)."""
    if mode == "uniform":
        return np.full(np.shape(theta_deg), NSEC, dtype=int)
    return (np.floor(np.mod(np.asarray(theta_deg) - 180.0, 360.0) / (360.0 / NSEC)).astype(int)) % NSEC


def centres(A):
    """Cell centres of a node array (ni, nj, nk)."""
    return 0.125 * (A[:-1, :-1, :-1] + A[1:, :-1, :-1] + A[:-1, 1:, :-1] + A[1:, 1:, :-1]
                    + A[:-1, :-1, 1:] + A[1:, :-1, 1:] + A[:-1, 1:, 1:] + A[1:, 1:, 1:])


def rows(a):
    return "\n".join("%.15E" % v for v in np.asarray(a).ravel(order="F")) + "\n"


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("mesh"); ap.add_argument("design"); ap.add_argument("mode", choices=("phases", "uniform"))
    ap.add_argument("--pamb", type=float, default=5000.0); ap.add_argument("--tamb", type=float, default=290.0)
    ap.add_argument("--tend", type=float, default=3.0e-3); ap.add_argument("--dumps", type=int, default=12)
    ap.add_argument("--cfl", type=float, default=0.4)
    a = ap.parse_args()
    mesh = os.path.abspath(a.mesh)
    summ = mesh.replace(".dat", "_summary.json")
    C = os.path.join(HERE, "CASE_3D", "wf_%s_%s" % (a.design, a.mode))
    assert os.path.basename(C).startswith("wf_")
    for sub in ("MESH", "INPUT", "OUTPUT", "bin"):
        os.makedirs(os.path.join(C, sub), exist_ok=True)

    def ln(src, dst):
        if os.path.islink(dst) or os.path.exists(dst):
            os.remove(dst)
        os.symlink(src, dst)
    ln(mesh, os.path.join(C, "MESH", "mesh.tec"))
    for f in KIT.files:
        ln(os.path.join(KIT.dir, f), os.path.join(C, "INPUT", f))
    PH = json.load(open(KITMOD.phases_file()))
    P = PH["phases"]
    global NSEC
    NSEC = len(P) - 1                                                          # the sectors (the last entry = the mean state)
    YP = [KIT.y(p["Y"]) for p in P]
    a_s = np.array([np.sqrt(p["gamma"] * p["R"] * p["T"]) for p in P])          # the sound speed of each inlet state
    u_s = np.array([p["mach"] for p in P]) * a_s                                 # axial, laboratory = frame
    zones = tecmesh.zone_index(mesh)
    # ---- the inlet: one 405 record per inlet cell of every Nozzle block, in the frame
    inlet, stat = {}, []
    for iz, z in enumerate(zones):
        if not z[0].startswith("Nozzle"):
            continue
        X, Y, Z = tecmesh.read_zone(mesh, zones, iz)
        yf = 0.25 * (Y[0, :-1, :-1] + Y[0, 1:, :-1] + Y[0, :-1, 1:] + Y[0, 1:, 1:])      # the f1 face centres (nj-1, nk-1)
        zf = 0.25 * (Z[0, :-1, :-1] + Z[0, 1:, :-1] + Z[0, :-1, 1:] + Z[0, 1:, 1:])
        r, th = np.hypot(yf, zf), np.arctan2(zf, yf)
        k = phase_of(np.degrees(th), a.mode)
        wth = OMEGA * r
        wx, wy, wz = u_s[k], -wth * np.sin(th), wth * np.cos(th)
        mrel = np.sqrt(wx ** 2 + wth ** 2) / a_s[k]
        alpha, beta = np.arctan2(wy, wx), np.arctan2(wz, np.hypot(wx, wy))
        recs = []
        nj, nk = r.shape
        for kk in range(nk):                                                   # k outer, j inner: the builder's order
            for jj in range(nj):
                p = P[k[jj, kk]]
                recs.append([float(mrel[jj, kk]), p["T"], p["p"], float(alpha[jj, kk]), float(beta[jj, kk])] + list(YP[k[jj, kk]]))
        inlet[str(iz + 1)] = recs
        stat.append((mrel.min(), mrel.max(), wth.min(), wth.max()))
    stat = np.array(stat)
    json.dump(inlet, open(os.path.join(C, "INPUT", "inlet405.json"), "w"))
    subprocess.run([PY, os.path.join(HERE, "build_bc_thor_nozzle.py"), mesh, summ, os.path.join(C, "INPUT", "bc.txt"), "--pamb", "%.3f" % a.pamb,
                    "--tamb", "%.3f" % a.tamb, "--far", "406", "--inlet405-file", os.path.join(C, "INPUT", "inlet405.json")], check=True,
                   stdout=open(os.path.join(C, "bc_build.log"), "w"), stderr=subprocess.STDOUT)
    # ---- the initial field, RELATIVE velocities: the channel = the inlet pattern extruded along x, the exhaust = air at
    # rest in the laboratory (so turning at +omega r in the frame)
    Rair = RU / (1.0 / np.sum(AIR / WSP))
    rho_a = a.pamb / (Rair * a.tamb)
    out = os.path.join(C, "INPUT", "ic.tec")
    with open(out + ".tmp", "w") as f:
        f.write(' VARIABLES ="x" "y" "z" ' + "".join('"rho(%d)"' % (q + 1) for q in range(KIT.n)) + ' "u" "v" "w" "p" "kappa" "omega" "T" "g" "R"\n')
        for iz, (name, ni, nj, nk, _) in enumerate(zones):
            X, Y, Z = tecmesh.read_zone(mesh, zones, iz)
            f.write(" ZONE  T = Block%d, I=%d, J=%d, K=%d, DATAPACKING=BLOCK, VARLOCATION=([1-3]=NODAL,[4-%d]=CELLCENTERED), SOLUTIONTIME=%.15E\n"
                    % (iz + 1, ni, nj, nk, KIT.n + 12, 0.0))
            for arr in (X, Y, Z):
                f.write(rows(arr))
            yc, zc = centres(Y), centres(Z)
            r, th = np.hypot(yc, zc), np.arctan2(zc, yc)
            wth = OMEGA * r
            one = np.ones_like(r)
            if name.startswith("Nozzle"):
                k = phase_of(np.degrees(th), a.mode)
                T = np.array([p["T"] for p in P])[k]; pr = np.array([p["p"] for p in P])[k]
                g = np.array([p["gamma"] for p in P])[k]; Yk = np.array(YP)[k]                          # (..., n species)
                Rm = RU / (1.0 / np.sum(Yk / WSP, axis=-1))
                rho = pr / (Rm * T)
                cells = [rho * Yk[..., s] for s in range(KIT.n)] + [u_s[k], -wth * np.sin(th), wth * np.cos(th), pr, TURB[0] * one, TURB[1] * one, T, g, Rm]
            else:
                cells = [rho_a * AIR[s] * one for s in range(KIT.n)] + [0.0 * one, -wth * np.sin(th), wth * np.cos(th), a.pamb * one, TURB[0] * one, TURB[1] * one,
                                                                    a.tamb * one, 1.4 * one, Rair * one]
            for c in cells:
                f.write(rows(c))
    os.replace(out + ".tmp", out)
    open(os.path.join(C, "input.ini"), "w").write(INI.format(tend=a.tend, sol_dtime=a.tend / a.dumps, omega=OMEGA, cfl=a.cfl))
    json.dump(dict(design=a.design, mode=a.mode, mesh=mesh, pamb=a.pamb, tamb=a.tamb, tend=a.tend, cfl=a.cfl, omega=OMEGA, f_wave=F_WAVE,
                   kit=KIT.dir, phases=KITMOD.phases_file(), P0A=PH.get("P0A"), nsec=NSEC,
                   axis=[-1.0, 0.0, 0.0], frozen=True, mrel_inlet=[float(stat[:, 0].min()), float(stat[:, 1].max())],
                   wtheta_inlet=[float(stat[:, 2].min()), float(stat[:, 3].max())]), open(os.path.join(C, "case.json"), "w"), indent=1)
    print("built %s: inlet relative Mach %.3f .. %.3f, frame speed at the inlet %.0f .. %.0f m/s; ic %.0f MB"
          % (C, stat[:, 0].min(), stat[:, 1].max(), stat[:, 2].min(), stat[:, 3].max(), os.path.getsize(out) / 1e6))


if __name__ == "__main__":
    main()
